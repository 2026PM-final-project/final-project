cmake_minimum_required(VERSION 3.16)
project(BomberLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bomber_core src/game.cpp src/scores.cpp)
target_include_directories(bomber_core PUBLIC src)
if(MSVC)
    target_compile_options(bomber_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(bomber_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(bomber_tests tests/test_game.cpp)
target_link_libraries(bomber_tests PRIVATE bomber_core)
enable_testing()
add_test(NAME rules_and_bonus COMMAND bomber_tests)
if(WIN32)
    add_executable(BombermanDemo WIN32 src/main.cpp src/renderer.cpp src/retro.cpp src/audio.cpp)
    target_link_libraries(BombermanDemo PRIVATE bomber_core opengl32 gdi32 winmm)
    add_executable(bomber_controls_tests tests/test_controls.cpp src/renderer.cpp src/retro.cpp src/audio.cpp)
    target_link_libraries(bomber_controls_tests PRIVATE bomber_core opengl32 gdi32 winmm)
    add_test(NAME window_controls COMMAND bomber_controls_tests)
    if(MSVC)
        target_compile_options(bomber_controls_tests PRIVATE /W4 /utf-8)
    elseif(MINGW)
        target_link_options(bomber_controls_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    if(MSVC)
        target_compile_options(BombermanDemo PRIVATE /W4 /utf-8)
    elseif(MINGW)
        target_link_options(BombermanDemo PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(bomber_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endif()
